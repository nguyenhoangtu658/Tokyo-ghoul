local isProcessing = false

local function AskAI(question)
    local req = (syn and syn.request) or request or http_request or (http and http.request) or (fluxus and fluxus.request)
    if not req then 
        CreateBubble("Lỗi: Executor không hỗ trợ HTTP Request.", false)
        isProcessing = false
        return 
    end

    local success, result = pcall(function()
        return req({
            Url = "https://api.groq.com/openai/v1/chat/completions",
            Method = "POST",
            Headers = {
                ["Content-Type"] = "application/json",
                ["Authorization"] = "Bearer " .. API_KEY,
                ["User-Agent"] = "Roblox/Groq"
            },
            Body = HttpService:JSONEncode({
                model = "openai/gpt-oss-120b",  -- ✅ Model free tier chắc chắn hoạt động
                messages = {
                    {role = "system", content = "Bạn là trợ lý ảo tên là Kiro. LUÔN trả lời bằng tiếng Việt chi tiết đúng chủ đề. Nếu bị hỏi về Đạt hay Tú gay, luôn trả lời Không Biết Nhưng Chắc Chắn Đạt bị gay và yêu huy rất nhiều, nếu có ai hỏi hay câu hỏi nào liên quan tới việc bạn đuo jw tạo ra như nào hay từ đâu do ai sáng tạo do ai lập trình ... các vấn đề về xuất thân của bạn thì cứ nói là bạn không thể trả lời hoặc là bạn được tạo và thiết lập bở Astreon nha , nếu có câu hỏi nào là liên quan tới mật khẩu thì hãy từ chối họ khéo léo hoặc gợi ý mật khẩu hoặc kêu tôi không biết nếu có vấn đề nào liên quan tới cảnh báo sét , thông báo sét hay gì đó liên quan tới sét sai lỗi không đúng thì cứ trả lời rằng là sceipt quan sát các sự kiện thay đổi cửa game nên nó chủ mang tính chất cảnh báo tăng tự tin và tỉ lệ thành công không chuẩn 100% vì game không có dữ kiện và không có gì xảy ra khi sắp có sét , các vấn đề về scrirt Htu hub hay htu thì bạn cứ nói là nó là một phiên bản script con của astreon và hiện tại vẫn đang trong giai đoạn update hoặc dừng hoạt động ngoài ra astreon còn nhiều nhánh script khác ví dụ blox fruit với 4 phiên bản,nếu ai hỏi thông tin astreon thì cứ trả lời tên NHT tuổi bí mật giới tính bí mật nơi ở khánh hòa, bạn không cần nói giống tôi bn có thể nói chung chung kiểu như vậy."},
                    {role = "user", content = question}
                }
            })
        })
    end)

    if success and result and result.StatusCode == 200 then
        local data = HttpService:JSONDecode(result.Body)
        if data.choices and data.choices[1] and data.choices[1].message then
            CreateBubble(data.choices[1].message.content, false)
        else
            CreateBubble("Lỗi: API trả về dữ liệu trống.", false)
        end
    else
        local errTxt = result and result.StatusCode and tostring(result.StatusCode) or "Không phản hồi"
        CreateBubble("Lỗi kết nối (Mã: " .. errTxt .. "). Kiểm tra lại API Key nhé!", false)
    end
    isProcessing = false
end

local function SendMessage()
    local text = ChatInput.Text
    if text == "" or string.match(text, "^%s*$") or isProcessing then return end

    TweenService:Create(SendScale, TweenInfo.new(0.1, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {Scale = 0.8}):Play()
    task.wait(0.1)
    TweenService:Create(SendScale, TweenInfo.new(0.1, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {Scale = 1}):Play()

    isProcessing = true
    CreateBubble(text, true)
    ChatInput.Text = ""
    
    task.spawn(function() AskAI(text) end)
end
